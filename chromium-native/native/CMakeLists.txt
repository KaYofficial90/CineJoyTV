cmake_minimum_required(VERSION 3.16)
project(multilogin_identity LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(multilogin_identity STATIC profile_identity.cc)
target_include_directories(multilogin_identity PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(identity_tests tests.cc)
target_link_libraries(identity_tests PRIVATE multilogin_identity)
enable_testing()
add_test(NAME identity_tests COMMAND identity_tests)
